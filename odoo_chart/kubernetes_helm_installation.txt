K3s (Lightweight Kubernetes)
=================================
Ubuntu Update
==================
sudo apt update
sudo apt upgrade -y

K3s Install
===============
curl -sfL https://get.k3s.io | sh -
sudo systemctl status k3s

kubectl Check
===============
sudo kubectl get nodes

kubectl Alias
=============
echo 'alias k=kubectl' >> ~/.bashrc
source ~/.bashrc
k get nodes

Cluster Info
=======================
kubectl cluster-info

Test Pod
===============
kubectl run nginx \
  --image=nginx \
  --port=80

kubectl get pods

Create Deployment
======================
kubectl create deployment nginx \
  --image=nginx

kubectl get deployments

Create Service
==================
kubectl expose deployment nginx \
  --port=80 \
  --type=NodePort

kubectl get svc

Access
===========
kubectl get svc
http://SERVER_IP:32000

Install Helm
================
curl -fsSL \
https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 \
| bash

helm version

Install Nginx Ingress
=========================
kubectl apply -f \
https://raw.githubusercontent.com/kubernetes/ingress-nginx/main/deploy/static/provider/cloud/deploy.yaml

kubectl get pods -n ingress-nginx

Install Metrics Server
========================
kubectl apply -f \
https://github.com/kubernetes-sigs/metrics-server/releases/latest/download/components.yaml

odoo create
===============
helm install odoo . --create-namespace -n odoo

sudo kubectl get all -n odoo

Upgrade
============
helm upgrade odoo . -n odoo

Check
========
kubectl get ingress -n odoo
kubectl describe ingress -n odoo

odoo run
==========
kubectl get pods -n odoo

odoo/postgres -service
=======================
kubectl get svc -n odoo

odoo-ingress
==============
kubectl get ingress -n odoo

New
======
kubectl get pods -n odoo -w

Check
=========
kubectl get pvc -n odoo

Odoo Deployment Restart
=================================
kubectl rollout restart deployment odoo -n odoo

Container Service
========================
kubectl exec -it -n odoo deployment/odoo -- bash
ping postgres-service

Deployment Check
=======================
kubectl get deployment -n odoo


Database Uninstall
===================
helm uninstall odoo -n odoo
kubectl delete pvc postgres-pvc -n odoo
kubectl delete namespace odoo
kubectl delete pod -n odoo -l app=postgres
helm install odoo . -n odoo










kubectl exec -it -n odoo deploy/postgres -- psql -U odoo -d postgres -c '\l'
